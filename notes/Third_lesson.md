# NumPy: автозаполнение, матрицы, диапазоны

## Автозаполнение

np.empty((3, 2))          # пустой массив (мусор)
np.zeros((2, 3, 4))       # массив из нулей
np.ones([4, 3])           # массив из единиц
np.full((3, 2), -1)       # массив из числа -1

## Матрицы

np.eye(4)                 # единичная матрица 4x4
np.eye(4, 2)              # единичная матрица 4x2
np.identity(4)            # единичная матрица 4x4
np.diag([1, 2, 3])        # диагональная матрица
np.diagflat([(1, 2), (3, 4)])  # плоская диагональ
np.tri(4, 2)              # треугольная матрица
np.tril([...])            # нижняя треугольная
np.triu([...])            # верхняя треугольная
np.vander([1, 2, 3])      # матрица Вандермонда

## Диапазоны

np.arange(5)              # 0..4
np.arange(0, 10, 2)       # 0, 2, 4, 6, 8
np.linspace(0, 1, 5)      # 0, 0.25, 0.5, 0.75, 1
np.logspace(0, 2, 3)      # 1, 10, 100
np.geomspace(1, 8, 4)     # 1, 2, 4, 8

## Что запомнить

zeros, ones, full — автозаполнение;
eye, identity — единичная матрица;
arange, linspace — диапазоны;
diag, tri — матрицы.

Остальное — гуглить по необходимости.
